% Flush the previous problem BEFORE updating its metadata and page marks.
\newcommand{\problemheader}[2]{%
    \clearpage
    \setproblem{#1}{#2}%
    \markboth{Problem #1}{Problem #1}%
    \phantomsection
    \addcontentsline{toc}{section}{#1. #2}%
    \begin{center}
        {\Large\bfseries Problem #1\par}
        \vspace{0.4em}
        {\LARGE\bfseries #2\par}
    \end{center}
}

% Empty limits are hidden while a statement is being drafted.
\newcommand{\problemlimits}[2]{%
    \ifstrempty{#1}{\ifstrempty{#2}{}{\begin{center}메모리 제한: #2\end{center}}}{%
        \begin{center}
            시간 제한: #1%
            \ifstrempty{#2}{}{\quad 메모리 제한: #2}%
        \end{center}
    }%
}
